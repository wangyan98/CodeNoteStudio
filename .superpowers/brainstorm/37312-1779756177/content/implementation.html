New IPC Handlers

// dialog:select-folder — native folder picker
ipcMain.handle('dialog:select-folder', async () => {
  const { dialog } = require('electron')
  const result = await dialog.showOpenDialog({
    properties: ['openDirectory']
  })
  return result.canceled ? null : result.filePaths[0]
})

// workspace:open — switch to a new workspace
ipcMain.handle('workspace:open', async (_event, newPath: string) => {
  currentProjectPath = newPath
  await saveLastWorkspacePath(newPath)    // persist to userData
  initDatabase(newPath)                   // re-init SQLite
  return loadConfig(newPath)              // return notebook.json
})

// workspace:get-current
ipcMain.handle('workspace:get-current', () => currentProjectPath)
